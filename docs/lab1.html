g_elist_from_alist <- g_adj_list %>% 
  select(id, friend1:friend2) %>% # keep only the network variables (and id)

g_elist_from_alist <- g_elist_from_alist%>%

library(igraph)

2.1 load data:

PupilsWaveV <- read_dta(file = "PupilsWaveV.dta")
class12b <- PupilsWaveV %>% 
  filter(schoolnr == "12b")  # keep only one class

2.2 select variables and into igraph:

class12b_clean <- class12b %>%
  select(namenr, age, sex, friend1:friend12)
  

class12b_elist <- class12b_clean %>%
  select(namenr, friend1:friend12) %>%
  melt(id.vars = "namenr") %>%
  filter(!is.na(value)) %>%
  rename(from = "namenr", to = "value", sourcevar = "variable") %>%
  relocate(to, .after = from)

2.3 this is a directed network, but we can check using R:

g_class12b <- graph_from_data_frame(class12b_elist, vertices = class12b_clean)

2.6 test the hypothesis that girls have higher degree than boys in this network

degrees <- degree(g_class12b, mode = "all")

node_data <- data.frame(
  namenr = V(g_class12b)$name,
  sex = V(g_class12b)$sex,
  degree = degrees
)
boxplot(degree ~ sex, 
        data = node_data, 
        main = "Degree Distribution by Gender",
        xlab = "Sex", 
        ylab = "Degree",
        col = c("lightblue", "pink"))

t test:

t_test_result <- t.test(degree ~ sex, data = node_data)
print(t_test_result)
#>
